cmake_minimum_required(VERSION 3.16)
project(Denoise LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HW5_SANITIZE "Enable ASan/UBSan" OFF)
find_package(OpenMP REQUIRED)
find_package(ZLIB REQUIRED)
add_library(denoiser src/denoiser.cpp src/util/mathutil.cpp)
target_include_directories(denoiser PUBLIC src src/ext)
target_link_libraries(denoiser PUBLIC OpenMP::OpenMP_CXX)
target_compile_options(denoiser PRIVATE -Wall -Wextra)
if(HW5_SANITIZE)
  target_compile_options(denoiser PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(denoiser PUBLIC -fsanitize=address,undefined -no-pie)
endif()
add_executable(Denoise src/main.cpp src/util/image.cpp src/util/imageutil.cpp)
target_link_libraries(Denoise PRIVATE denoiser ZLIB::ZLIB)
add_executable(denoiser-tests tests/denoiser-tests.cpp)
target_link_libraries(denoiser-tests PRIVATE denoiser)
enable_testing()
add_test(NAME analytic COMMAND denoiser-tests)

add_executable(io-tests tests/io-tests.cpp src/util/image.cpp src/util/imageutil.cpp)
target_link_libraries(io-tests PRIVATE denoiser ZLIB::ZLIB)
